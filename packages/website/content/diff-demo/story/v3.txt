export const consequat = (duis: Sit, aute: number) => {
    const irure = {
        lorem: duis.lorem,
        ipsum: duis.ipsum,
        dolor: duis.dolor,
        amet: duis.amet,
    };
    if (aute > LOREM) {
        const reprehenderit = dolor(irure, aute);

    return irure;
};

export const voluptate = (velit: Sit) => {
    const esse = velit.lorem === IPSUM ? velit.ipsum : velit.dolor;
