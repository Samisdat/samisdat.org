```typescript
const doParallax = () => {
    const element = ref.current;
    const { x, y } = coordsRef.current;

    element.style.setProperty('--parallax-x', String(x));

    element.style.setProperty('--parallax-y', String(y));
};

useAnimationFrame(doParallax);
```

```css
.parallax {
    transition: transform 10ms ease;
    transform: translateX(calc(150px * var(--parallax-x) * var(--depth)))
        translateY(calc(150px * var(--parallax-y) * var(--depth)));
}
```
